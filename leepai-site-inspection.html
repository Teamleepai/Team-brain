The host is Cloudflare. Serve a real 404 status for unmatched routes. If the site is on Cloudflare Pages, add a _routes.json and a Function; the portable fix that works on any Cloudflare zone is a Worker in front of the asset handler: 
// functions/_middleware.js  (Cloudflare Pages Functions)
const KNOWN = new Set(['/', '/demo', '/privacy', '/terms', '/sms-terms']);

export async function onRequest(context) {
  const { request, next } = context;
  const url = new URL(request.url);
  const path = url.pathname.replace(/\/+$/, '') || '/';

  const isAsset = /\.[a-z0-9]+$/i.test(path);
  if (isAsset || KNOWN.has(path)) return next();

  // Unmatched route: serve the SPA shell but with a real 404 status.
  const shell = await next(new Request(new URL('/', url), request));
  return new Response(shell.body, {
    status: 404,
    headers: shell.headers,
  });
}

Keep the client-side 404 view — it is good UX. This change only corrects the status code on the wire, which is the part Google reads.
